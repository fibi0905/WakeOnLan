import express from "express";
import cors from "cors";
import dgram from "dgram";
import fs from "fs";
import path from "path";
import { execSync } from 'child_process';
import * as net from 'net';
import e from "express";

const app = express();
app.use(cors());
app.use(express.json());

// ── File JSON ────────────────────────────────────────────────
const DATA_PATH = path.join(__dirname, "../data/pc.json");

if (!fs.existsSync(path.dirname(DATA_PATH))) {
  fs.mkdirSync(path.dirname(DATA_PATH), { recursive: true });
}
if (!fs.existsSync(DATA_PATH)) {
  fs.writeFileSync(DATA_PATH, "[]", "utf-8");
}

// ── Tipi ────────────────────────────────────────────────────
type Pc = { nome: string; mac: string; };

const readPcs  = (): Pc[] => JSON.parse(fs.readFileSync(DATA_PATH, "utf-8"));
const writePcs = (pcs: Pc[]) =>
  fs.writeFileSync(DATA_PATH, JSON.stringify(pcs, null, 2), "utf-8");

// ── GET /pcs — lista tutti i PC ──────────────────────────────
app.get("/pcs", (req, res) => {
  res.json(readPcs());
});

// ── POST /pcs — aggiunge un PC ──────────────────────────────
app.post("/pcs", (req, res) => {
  const { nome, mac } = req.body;
  if (!nome || !mac)
    return res.status(400).json({ error: "nome e mac sono obbligatori" });

  const pcs = readPcs();
  if (pcs.find(p => p.nome === nome))
    return res.status(409).json({ error: "Nome già esistente" });

  pcs.push({ nome, mac });
  writePcs(pcs);
  res.status(201).json({ nome, mac });
});

// ── DELETE /pcs/:nome — elimina un PC ───────────────────────
app.delete("/pcs/:nome", (req, res) => {
  const pcs = readPcs();
  const filtrati = pcs.filter(p => p.nome !== req.params.nome);
  if (filtrati.length === pcs.length)
    return res.status(404).json({ error: "PC non trovato" });

  writePcs(filtrati);
  res.json({ success: true });
});

// ── POST /wol — invia magic packet ──────────────────────────
function sendWakeOnLan(mac: string): Promise<void> {
  return new Promise((resolve, reject) => {
    const macClean = mac.replace(/[:\-]/g, '').toUpperCase();
    if (macClean.length !== 12) return reject(new Error("MAC non valido"));

    const macBytes = Buffer.from(macClean, "hex");
    const packet   = Buffer.alloc(102);
    packet.fill(0xff, 0, 6);
    for (let i = 1; i <= 16; i++) macBytes.copy(packet, i * 6);

    const socket = dgram.createSocket("udp4");
    socket.bind(() => {
      socket.setBroadcast(true);
      socket.send(packet, 0, packet.length, 9, "255.255.255.255",
        (err) => { socket.close(); err ? reject(err) : resolve(); });
    });
  });
}


app.post("/wol", async (req, res) =>{
  const { name } = req.body;
  if (!name) return res.status(400).json({ error: "nome obbligatorio" });  
  const pcs = readPcs();

  const pc = pcs.find(p => p.nome === name)

  if(pc){
    try {
      await sendWakeOnLan(pc.mac);
      res.json({ success: true });
    } catch (err) {
      res.status(500).json({ success: false, error: String(err) });
    }
  }
  else{
    return res.status(400).json({ error: "nome non esitente" });
  }


});

// ── GET IP da MAC ────────────────────────────────────────────
function getIp(mac: string): string | null {
    try {
        const macNormalized = mac.replace(/[:\-]/g, '').toLowerCase();
        const content = fs.readFileSync('/proc/net/arp', 'utf-8');

        for (const line of content.split('\n').slice(1)) {
            const parts = line.trim().split(/\s+/);
            if (parts.length < 4) continue;

            const lineMac = parts[3].replace(/:/g, '').toLowerCase();
            if (lineMac === macNormalized) return parts[0];
        }

        return null;
    } catch (error) {
        console.error("Errore in getIp:", error);
        return null;
    }
}

// ── Invia messaggio TCP ──────────────────────────────────────
function sendMsg(ip: string, port: number, msg: string): Promise<void> {
    return new Promise((resolve, reject) => {
        const client = new net.Socket();

        client.setTimeout(3000);

        client.connect(port, ip, () => {
            client.write(msg, (err) => {
                if (err) return reject(err);
                client.end(() => resolve());
            });
        });

        client.on('error', (err) => {
            reject(err);
        });

        client.on('timeout', () => {
            client.destroy();
            reject(new Error('Timeout'));
        });
    });
}

app.post("/down", async (req, res) => {
  const { name } = req.body;
  if (!name) return res.status(400).json({ error: "nome obbligatorio" });  
  const pcs = readPcs();

  const pc = pcs.find(p => p.nome === name)

  if(pc){

import React from 'react';
import { IonAlert, IonButton, IonContent, IonFabButton, IonHeader, IonIcon, IonInput, IonItem, IonLabel, IonPage, IonSpinner, IonText, IonTitle, IonToolbar } from '@ionic/react';
import './Log.css';
import { RouteComponentProps, withRouter } from 'react-router-dom';


import { logIn} from '../../Typescript/pcService';
import lock from '../../theme/Icon/lock-closed-outline.svg'
import close from '../../theme/Icon/close-outline.svg'
import arrow from '../../theme/Icon/arrow-forward-outline.svg'
import isMobile from 'is-mobile';

type OwnProps = {
  rederict: ()=> void,
};
type Props = OwnProps & RouteComponentProps;

type State = {
  loading:boolean,
  correct:boolean,
}

class Log extends React.Component<Props, State> {

  constructor(props: any) {
    super(props);
    this.state = {
      loading:false,
      correct:true,

    }
  }

  pass = React.createRef<HTMLIonInputElement>()

  changeState = (ok:boolean)=>{
    if(ok){
      this.setState({
        loading:false,
      })
    }
    else{
      this.setState({
        loading:false,
        correct:false,
      })

      setTimeout(()=>{
        this.setState({
          correct:true,
        })
      }, 2000);
    }
  }

  log = async()=>{
    if(this.pass.current ){
      let passIn = this.pass.current.value as string

      this.pass.current.value = ""
      
      if(await logIn(passIn)){
        this.changeState(true)
        this.props.rederict()
        this.props.history.push('/home')
      }
      else{
        this.changeState(false)
      }
      
    }
    else{
      this.changeState(false)
    }
  }

  

  render() {
    const {loading} = this.state;
    return (
      <IonPage >
        <IonHeader className='header'>
          <div className="headerCont">
            <IonText className='logIN'>Login</IonText>
          </div>

        </IonHeader>
        <IonContent scrollX={false} scrollY={false}>
          <div className="mainCont">

            <div className={`formLoginGrid${isMobile() ? ' formGridMobile' : ''}`}>
              <div className="iconFromC">

                {!loading &&

                  <IonIcon
                    className={`iconFrom${!this.state.correct ? ' incorect' : ''}`}
                    icon={
                      this.state.correct ?
                        lock
                      :
                        close
                      
                    }
                  ></IonIcon>
                }

                {loading &&
                  <IonSpinner 
                    name="lines-sharp"
                    className='spiner'
                  ></IonSpinner>
                
                }
              </div>

              <div className="inputTextC">
                <IonInput
                  className='inputText'
                  required={true}
                  clearOnEdit={true}  
                  fill="outline"
                  mode='ios'
                  type='password'
                  placeholder='Password' 
                  onKeyUp={(e: any) => {
                    if (e.key === 'Enter' && !isMobile()) {
                      this.log();
                    }
                  }}
                  ref={this.pass}

                ></IonInput>

              </div>

              <div className="sendC">
                <IonFabButton 
                  className='send'
                  onClick={()=>{
                   this.log()
                  }}
                >
                  <IonIcon
                    icon={arrow}
                  ></IonIcon>
                </IonFabButton>

              </div>



            </div>

          </div>
        </IonContent>
      </IonPage>
    );
  }
}

import React from 'react';
import { IonAlert, IonButton, IonIcon, IonLabel } from '@ionic/react';
import './PcCard.css';

import pc from '../../theme/Icon/desktop-outline.svg'
import option from '../../theme/Icon/ellipsis-horizontal-outline.svg'
import close from '../../theme/Icon/close-outline.svg'
import power from '../../theme/Icon/power-outline.svg'
import trash from '../../theme/Icon/trash-outline.svg'

import { deletePc, wakePC, shutdown } from '../../Typescript/pcService';
import isMobile from 'is-mobile';


type Props = {
  Name: string,
  MAC: string
  onDelete: () =>void
}

type State = {
  open: boolean,
  work: boolean,
  confirm: boolean,
  del:boolean,
  on: boolean,
  off: boolean

}

class PcCard extends React.Component<Props, State> {

  constructor(props: any) {
    super(props);
    this.state = {
      open: false,
      work: false,
      confirm: false,
      del: false,
      on:false,
      off:false,

    }
  }


  handleDelete = async () => {
    await deletePc(this.props.Name as string);
    this.props.onDelete(); // ← avvisa Home di ricaricare
  }

  on = async ()=>{
    await wakePC(this.props.Name as string);
    this.setState({on: true});
    
    setTimeout( ()=>{
      this.setState({on: false});
      
    }, 1500);
    
  }

  off = async ()=>{
    await shutdown (this.props.Name as string);
    this.setState({off: true});
    
    setTimeout( ()=>{
      this.setState({off: false});
      
    }, 1500);
  }

  render() {
    const { open } = this.state;

    return (
      <>
        <IonAlert 
          isOpen ={this.state.confirm}
          header= 'Vuoi cancellare ?'
          message={`${this.props.Name}`}
          buttons={[
            {
              text: 'Si',
              handler: () =>{
                this.setState({confirm:false});
                this.handleDelete();
              }
              
            },
            {
              text: 'NO',
              handler: ()=>{
                this.setState({confirm:false});
              }
            }
          ]}  
        >

        </IonAlert>
      
        <div className={`CardGrid${open ? ' open' : ''}`}>

          {/* Prima riga — non cambia mai grid-area, niente scatti */}

          <div className={`IconPc${isMobile() ? ' IcPcMobile' : ''}`}>
            <IonIcon className={`icon${isMobile() ? ' icoMobile' : ''}`} icon={pc} />
          </div>

          <div className='Label'>
            <IonLabel class='label'>{this.props.Name}</IonLabel>
          </div>

          <div
            className={`Option${isMobile() ? ' OptionMobile' : ''}`}
            onClick={() => this.setState({ open: !open })}
          >
            <IonIcon className={`icon${isMobile() ? ' icoMobile' : ''}`} icon={open ? close : option} />
          </div>

          {/* Seconda riga — contenuto espanso, sempre nel DOM ma nascosto */}
          <div className='CardContent'>
            <div 
              className= {`Botton${isMobile() ? ' mobileB' : ''} power${this.state.on ? ' on' : ''}`}
              onClick={()=>{this.on();}}
            >
              <IonLabel>{this.state.on ? 'Avviato' : 'Avvia' }</IonLabel>
             

              {!this.state.on &&
                <IonIcon icon={power}></IonIcon>
              }
             
             
            </div>

            <div 
              className= {`Botton${isMobile() ? ' mobileB' : ''} power${!this.state.off ? ' off' : ' active'}`}
              onClick={()=>{this.off();}}
            >
              <IonLabel>{this.state.off ? 'Spento' : 'Spegni' }</IonLabel>
      
              {!this.state.off &&
                <IonIcon icon={power}></IonIcon>
              }
            
            
            </div>


            <div 
              className={`Botton delete ${isMobile() ? ' mobileB' : ''} `}
              onClick={()=>{this.setState({confirm:true})}}
            >
              <IonLabel>Cancella</IonLabel>
              <IonIcon icon={trash}></IonIcon>
            </div>


          </div>

        </div>
      </>
    );
  }
}

export default PcCard;

const BASE_URL = 'https://apiwol.cloudnic.cv';

export type Pc = { nome: string; mac: string; };

export const getPcs = async (): Promise<Pc[]> => {
  const res = await fetch(`${BASE_URL}/pcs`);
  return res.json();
};

export const addPc = async (nome: string, mac: string): Promise<Pc> => {
  const res = await fetch(`${BASE_URL}/pcs`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ nome, mac }),
  });
  return res.json();
};

export const deletePc  = async (nome: string): Promise<void> => {
  await fetch(`${BASE_URL}/pcs/${encodeURIComponent(nome)}`, { method: 'DELETE' });
};

export const wakePC = async (name: string): Promise<void> => {
  await fetch(`${BASE_URL}/wol`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ name }),
  });
};

export const shutdown = async (name:string): Promise<void> =>{
  await fetch(`${BASE_URL}/down`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ name }),
  });
}


export const logIn = async (pass:string): Promise<boolean> =>{
  const rest = await fetch(`${BASE_URL}/log`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ pass }),
  })

}
